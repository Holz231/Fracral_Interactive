cmake_minimum_required(VERSION 3.20)
project(AuroraFractal VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/fractal.hlsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/fractal.hlsl" FRACTAL_SHADER)
# MSVC limits each literal to 16 KiB; adjacent literals retain one embedded shader.
string(REPLACE "\nfloat" ")AURORASHADER\"\nR\"AURORASHADER(\nfloat" FRACTAL_SHADER "${FRACTAL_SHADER}")
configure_file(src/shader_data.h.in generated/shader_data.h @ONLY)
add_executable(AuroraFractal WIN32 src/main.cpp src/renderer.cpp)
target_include_directories(AuroraFractal PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(AuroraFractal PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(AuroraFractal PRIVATE d3d11 dxgi d3dcompiler d2d1 dwrite windowscodecs ole32 user32 gdi32 dwmapi shell32)
if(MSVC)
  target_compile_options(AuroraFractal PRIVATE /W4 /O2 /EHsc /utf-8)
  target_link_options(AuroraFractal PRIVATE /DYNAMICBASE /NXCOMPAT)
endif()
add_executable(FractalTests tests/world_tests.cpp)
target_include_directories(FractalTests PRIVATE src)
if(MSVC)
  target_compile_options(FractalTests PRIVATE /W4 /O2 /EHsc)
endif()
enable_testing()
add_executable(CitadelTests tests/citadel_tests.cpp)
target_include_directories(CitadelTests PRIVATE src)
if(MSVC)
  target_compile_options(CitadelTests PRIVATE /W4 /O2 /EHsc)
endif()
add_test(NAME citadel_continuity COMMAND CitadelTests)
add_test(NAME fractal_geometry COMMAND FractalTests)
add_executable(FrameSettlingTests tests/frame_settling_tests.cpp)
target_include_directories(FrameSettlingTests PRIVATE src)
add_test(NAME frame_settling COMMAND FrameSettlingTests)
add_executable(RaceTests tests/race_tests.cpp)
target_include_directories(RaceTests PRIVATE src)
target_compile_definitions(RaceTests PRIVATE NOMINMAX)
add_test(NAME race_ghost COMMAND RaceTests)
add_executable(RendererTests tests/renderer_tests.cpp src/renderer.cpp)
target_include_directories(RendererTests PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(RendererTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(RendererTests PRIVATE d3d11 dxgi d3dcompiler d2d1 dwrite windowscodecs ole32 user32 gdi32 dwmapi shell32)
if(MSVC)
  target_compile_options(FrameSettlingTests PRIVATE /W4 /O2 /EHsc)
  target_compile_options(RaceTests PRIVATE /W4 /O2 /EHsc)
  target_compile_options(RendererTests PRIVATE /W4 /O2 /EHsc /utf-8)
endif()
add_test(NAME renderer_state COMMAND RendererTests)
set_tests_properties(renderer_state PROPERTIES LABELS gpu RUN_SERIAL TRUE TIMEOUT 180)
install(TARGETS AuroraFractal RUNTIME DESTINATION .)
